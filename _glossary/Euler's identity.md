$$\begin{aligned} e^{j\theta} = \cos(\theta) + j \sin(\theta)\\
 e^{-j\theta} = \cos(\theta) - j \sin(\theta) \\  \\
 \cos(\theta)=\frac{1}{2}(e^{j\theta}+e^{-j\theta})\\
 \sin(\theta)=\frac{1}{2j}(e^{j\theta}-e^{-j\theta})
\end{aligned}$$
